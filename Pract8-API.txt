
# Weather Data Analysis using OpenWeatherMap API

import requests
import pandas as pd
import matplotlib.pyplot as plt

# 1. API Key and Multiple Cities
api_key = "9aa42e4d2bdf42dccc48675b992968cc"

cities = ["Pune", "Mumbai", "Delhi"]

# 2. Get Weather Data from API
url = "https://api.openweathermap.org/data/2.5/weather"

weather_list = []

for city in cities:

    params = {
        "q": city,
        "appid": api_key,
        "units": "metric"
    }

    try:
        response = requests.get(url, params=params, timeout=15)

        print("Status Code:", response.status_code)

        response.raise_for_status()
        data = response.json()

        # 3. Extract Weather Information
        weather = {
            "City": city,
            "Temperature": data["main"]["temp"],
            "Humidity": data["main"]["humidity"],
            "Wind Speed": data["wind"]["speed"],
            "Pressure": data["main"]["pressure"]
        }

        weather_list.append(weather)

    except requests.exceptions.HTTPError:
        print("API Error for", city, ":", response.text)

    except requests.exceptions.RequestException as e:
        print("Connection Error for", city, ":", e)

    except KeyError:
        print("Weather data missing for", city)


# 4. Create DataFrame
df = pd.DataFrame(weather_list)

if df.empty:
    print("No weather data available. Check your API key and internet connection.")

else:

    print("\nWeather Information:")
    print(df)

    # 5. Check and Clean Data
    print("\nNull Values:")
    print(df.isnull().sum())

    df = df.dropna()

    # 6. Weather Analysis
    print("\nAverage Temperature:", df["Temperature"].mean())
    print("Maximum Temperature:", df["Temperature"].max())
    print("Minimum Temperature:", df["Temperature"].min())
    print("Average Humidity:", df["Humidity"].mean())
    print("Average Wind Speed:", df["Wind Speed"].mean())

    # 7. Visualization

    # Temperature
    plt.bar(df["City"], df["Temperature"])
    plt.xlabel("City")
    plt.ylabel("Temperature (°C)")
    plt.title("Temperature Comparison")
    plt.show()

    # Humidity
    plt.bar(df["City"], df["Humidity"])
    plt.xlabel("City")
    plt.ylabel("Humidity (%)")
    plt.title("Humidity Comparison")
    plt.show()

    # Wind Speed
    plt.bar(df["City"], df["Wind Speed"])
    plt.xlabel("City")
    plt.ylabel("Wind Speed (m/s)")
    plt.title("Wind Speed Comparison")
    plt.show()

    # 8. Correlation
    correlation = df[
        ["Temperature", "Humidity", "Wind Speed"]
    ].corr()

    print("\nCorrelation:")
    print(correlation)

    plt.imshow(correlation, cmap="coolwarm", vmin=-1, vmax=1)

    plt.colorbar()
    plt.xticks(range(3), ["Temperature", "Humidity", "Wind Speed"])
    plt.yticks(range(3), ["Temperature", "Humidity", "Wind Speed"])
    plt.title("Weather Correlation")
    plt.show()

What is correlation?

Correlation measures the relationship between two variables.

Correlation values range from -1 to +1.
Value	Meaning
+1	Perfect positive correlation
0	No linear correlation
-1	Perfect negative correlation

For example:
- Positive correlation: When one variable increases, the other tends to increase.
- Negative correlation: When one variable increases, the other tends to decrease.

Humidity is the amount of water vapour present in the air.

Q12. What is atmospheric pressure?

It is the pressure exerted by the weight of the atmosphere, commonly measured in hPa.

Metres per second (m/s) in this practical.
